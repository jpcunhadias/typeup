\begin{topic}{Topic 1: Linear algebra}
  \begin{description}
    \item[Inverse (2×2)] \(\begin{pmatrix} a & b \\ c & d \end{pmatrix}^{-1}
      = \frac{1}{ad - bc}\begin{pmatrix} d & -b \\ -c & a \end{pmatrix}\)
    \item[Eigenvalues] roots of \(\det(A - \lambda I) = 0\)
    \item[Trace, determinant] \(\operatorname{tr} A = \sum_i \lambda_i\),
      \(\det A = \prod_i \lambda_i\)
  \end{description}
\end{topic}

\begin{topic}{Topic 2: Probability}
  \(\E[aX + b] = a\E[X] + b\), \quad \(\Var(aX + b) = a^2 \Var(X)\)
  \[
    P(A \mid B) = \frac{P(B \mid A)\,P(A)}{P(B)}
  \]
\end{topic}

\begin{topic}{Common mistakes}
  \begin{itemize}
    \item Sign of \(-1\) when expanding \(-(a - b)\).
    \item \(\Var(X + Y) = \Var X + \Var Y\) only if independent.
  \end{itemize}
\end{topic}
